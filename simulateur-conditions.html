📚 Aide Conditions

                
            

            
if / else / else if / switch

        


        

            
            

                
                
                
                
                
                
            


            
            

                

                    
❓ if simple

                    

                        
Une seule condition

                        
let age = 20;
if (age >= 18) {
    console.log("Majeur");
}

                    

                    

                        
if ... else

                        
let age = 15;
if (age >= 18) {
    console.log("Majeur");
} else {
    console.log("Mineur");
}

                    

                    

                        💡 La condition entre parenthèses doit être vraie (true) pour exécuter le bloc.
                    

                

            


            
            

                

                    
🔁 else if (plusieurs cas)

                    

                        
Notes scolaires

                        
let note = 14;
if (note >= 16) {
    console.log("Tres bien");
} else if (note >= 12) {
    console.log("Bien");
} else if (note >= 10) {
    console.log("Passable");
} else {
    console.log("Insuffisant");
}

                    

                    

                        💡 Les conditions sont testées de haut en bas. Le premier true gagne, les autres sont ignorés.
                    

                

            


            
            

                

                    
⚖ Opérateurs de comparaison

                    

                        
===  // Egal (valeur ET type)
!==  // Different
>    // Superieur
>=   // Superieur ou egal
<    // Inferieur
<=   // Inferieur ou egal

                    

                    

                        💡 Utilisez toujours === (et non ==) pour comparer valeur ET type.
                    

                

            


            
            

                

                    
🔗 Opérateurs logiques

                    

                        
ET logique && (les deux vrais)

                        
let age = 25;
if (age >= 18 && age <= 65) {
    console.log("Actif");
}

                    

                    

                        
OU logique || (au moins un vrai)

                        
let jour = "samedi";
if (jour === "samedi" || jour === "dimanche") {
    console.log("Week-end");
}

                    

                

            


            
            

                

                    
⚡ Opérateur ternaire

                    

                        
condition ? siVrai : siFaux

                        
let age = 20;
let statut = age >= 18 ? "Majeur" : "Mineur";
console.log(statut); // Majeur

                    

                    

                        💡 Le ternaire est un raccourci de if/else qui renvoie une valeur.
                    

                

            


            
            

                

                    
🔀 switch

                    

                        
switch / case / break / default

                        
let jour = 3;
switch (jour) {
    case 1:
        console.log("Lundi");
        break;
    case 2:
        console.log("Mardi");
        break;
    case 3:
        console.log("Mercredi");
        break;
    default:
        console.log("Inconnu");
}

                    

                    

                        
Regrouper des cas

                        
switch (jour) {
    case "samedi":
    case "dimanche":
        console.log("Week-end");
        break;
    default:
        console.log("Semaine");
}

                    

                    

                        💡 N'oubliez pas break; après chaque cas, sinon l'exécution continue sur le cas suivant.
